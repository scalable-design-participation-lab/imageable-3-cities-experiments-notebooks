Define validation metrics

def compute_mse(ground_truths, predicted_values):

    mse = 0
    n_data = len(ground_truths)
    for i in range(n_data):
        y_truth = ground_truths[i]
        y_predicted = predicted_values[i]
        mse += ((y_truth - y_predicted)**2)/n_data

    return mse

def compute_rmse(ground_truths, predicted_values):
    mse = compute_mse(ground_truths, predicted_values)

    return np.sqrt(mse)


def compute_mae(ground_truths, predicted_values):
    mae = 0
    n_data = len(ground_truths)
    for i in range(n_data):
        y_truth = ground_truths[i]
        y_predicted = predicted_values[i]
        mae += np.abs(y_truth - y_predicted)/n_data

    return mae


def compute_improvement(ground_truths, predicted_values, training_data):
    n_data = len(ground_truths)
    mean_val = sum(training_data[i] for i in range(len(training_data))) / len(training_data)
    rmse_dummy = compute_rmse(ground_truths, [mean_val] * n_data)
    rmse_model = compute_rmse(ground_truths, predicted_values)
    return 1.0 - rmse_model / rmse_dummy

monterrey_dictionaries = []
def get_all_characteristics(training_instances, validation_instances, model, model_family, city_name, combined = False):
    result_dictionary = {}
    result_dictionary["Model Family"] = model_family
    result_dictionary["N"] = len(validation_instances)
    result_dictionary["Validation City"] = city_name
    if(not combined):
        result_dictionary["Training"] = "City"
    else:
        result_dictionary["Training"] = "Combined"
    

    features = [x["features"] for x in validation_instances]
    if(model_family == "Footprint XGBoost"):
        #The footprint model only uses the 13 footprint features, which come first.
        features = [x["features"][:13] for x in validation_instances]
    heights_validation = [x["real_height"] for x in validation_instances]
    heights_training = [x["real_height"] for x in training_instances]
    predictions = model.predict(np.array(features))

    rmse = compute_rmse(
        heights_validation,
        predictions
    )

    mae = compute_mae(
        heights_validation,
        predictions
    )

    improvement = compute_improvement(
        heights_validation,
        predictions,
        heights_training
    )

    result_dictionary["RMSE"] = rmse
    result_dictionary["MAE"] = mae
    result_dictionary["Improvement (%)"] = improvement*100

    return result_dictionary
