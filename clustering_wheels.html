Our wheel visualization summarizes the following properties per-cluster:

  1. Building height (ground truth)
  2. RMSE-gain (against the single XGBoost model).
  3. City that contributes the highest share of validation instances for each of the cluster + the percentage.
  4. Dominant facade material.
  5. Representative footprint and street view image.

#Define a function to expand the materials.
def expand_materials(gdf):
    #This would be a list of dictionaries.
    materials = list(gdf["material_percentages"])
    materials = [ast.literal_eval(x) for x in materials]

    ALL_MATERIALS = ["asphalt","concrete","metal","road_marking","fabric_leather","glass","plaster", "plastic","rubber","sand", "gravel","ceramic","cobblestone","brick","grass","wood","leaf","water","human_body","sky"]

    BASE_STR = "material_pct_"
    #These would be all the materials
    keys = ALL_MATERIALS

    for i in range(len(keys)):
        material_name = keys[i]

        column_name = BASE_STR+material_name
        column_values = []
        for m in materials:
            #Case where the dictionary is empty
            if(not m):
                column_values.append(None)
            #Case where there is a percentages option in m (need to check why this happens)
            elif(material_name not in m and "percentages" in m and material_name in m["percentages"]):
                column_values.append(m["percentages"][material_name])
            #Case where the dictionary just has materials: percentage.
            elif(material_name in m):
                column_values.append(m[material_name])
            else:
                print(m)
                print(material_name)

        gdf[column_name] = column_values

    return gdf

def compute_rmse(predicted, ground_truths):
    sum_square_differences = 0
    n = len(predicted)
    for i in range(0,n):
        sum_square_differences += (predicted[i] - ground_truths[i])**2
    
    sum_square_differences = sum_square_differences/n
    return np.sqrt(sum_square_differences)

def get_rmse_for_gdf(gdf, ground_truths, model_ctp, model_single):
    rows = [i for i in gdf.index if i in index_to_features]
    features = np.array([index_to_features[i] for i in rows])

    predictions_ctp = list(model_ctp.predict(features))
    predictions_single = list(model_single.predict(features))

    rmse_single = compute_rmse(predictions_single, ground_truths)
    rmse_ctp = compute_rmse(predictions_ctp, ground_truths)

    gain = 100*(rmse_single - rmse_ctp)/rmse_single

    return gain

Majority City

Calculate the city that contributes the highest share of buildings for each cluster and the percentage it contributes.

def compute_majority_city_and_percentage(cities_list):
    unique_categories = list(np.unique(cities_list))
    n_unique_categories = len(unique_categories)
    counts = np.zeros(n_unique_categories)
    #Iterate over the categories
    for i in range(len(cities_list)):
        city = cities_list[i]
        index = unique_categories.index(city)
        counts[index] += 1
    
    counts = np.array(counts)
    shares = (counts/len(cities_list))*100
    index_max = np.argmax(shares)
    majority_city = unique_categories[index_max]
    majority_city_share = shares[index_max]

    return majority_city, majority_city_share

        wheel_element = WheelElement(
            label,
            group_name = f"Cluster {i+1}",
            feature_name = "Material"
        )
        wheel_elements[key][i].append(wheel_element)

def get_distance_to_cluster(vector, cluster_centroid):
    n_features = len(vector)
    dst = 0
    for i in range(0,n_features):
        dst += (vector[i] - cluster_centroid[i])**2
    
    dst = np.sqrt(dst)
    return dst


indexes_array = {}
indexes_array[POOLED_KEY] = [0, 0, 0, 0, 0, 0]
indexes_array[MONTERREY_KEY] = [1, 0, 0]
indexes_array[BOSTON_KEY] = [0, 0, 0]
indexes_array[SF_KEY] = [0, 0, 0]



def get_building_image_and_footprint(
    key,
    cluster_index
):
    gdf = cluster_gdfs[key][cluster_index]
    geometries = list(gdf["geometry"])
    image_paths = list(gdf["image_path"])
    feature_vectors = [index_to_features[j][:-1] for j in gdf.index]
    cluster_centroid = cluster_centroids[key][cluster_index]

    index_k_nearest_neighbor = indexes_array[key][cluster_index]
    distances_from_centroid = [get_distance_to_cluster(x, cluster_centroid) for x in feature_vectors]

    k_th_index = np.argsort(distances_from_centroid)[index_k_nearest_neighbor]

    footprint = geometries[k_th_index]
    image = np.array(Image.open(image_paths[k_th_index]))

    return footprint, image
